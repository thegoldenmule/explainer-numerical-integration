/// Which store something lives in; only `MergedLibrary` ever asks.
public enum LibraryProviderID: String, Sendable, Hashable, CaseIterable {
    /// Ours, always live — the feature works with Apple Music absent.
    case cloudKit

    /// The real library; live when `isSharedDevice` is false, the default.
    case appleMusic
}

/// What a provider can do; absence isn't failure, just sitting one out.
public struct LibraryCapabilities: OptionSet, Sendable, Hashable {
    public let rawValue: Int

    public init(rawValue: Int) {
        self.rawValue = rawValue
    }

    public static let favoriteSongs = LibraryCapabilities(rawValue: 1 << 0)
    public static let favoriteAlbums = LibraryCapabilities(rawValue: 1 << 1)
    public static let favoriteArtists = LibraryCapabilities(rawValue: 1 << 2)
    public static let createPlaylists = LibraryCapabilities(rawValue: 1 << 3)
    public static let editOwnPlaylists = LibraryCapabilities(rawValue: 1 << 4)
    public static let editForeignPlaylists = LibraryCapabilities(rawValue: 1 << 5)

    /// Everything our store can do — the bar the Apple provider must meet.
    public static let all: LibraryCapabilities = [
        .favoriteSongs, .favoriteAlbums, .favoriteArtists,
        .createPlaylists, .editOwnPlaylists, .editForeignPlaylists,
    ]

    /// The flag for one favorite kind; nil for a playlist, which is owned.
    public static func favoriting(_ kind: ItemKind) -> LibraryCapabilities? {
        switch kind {
        case .song: .favoriteSongs
        case .album: .favoriteAlbums
        case .artist: .favoriteArtists
        case .playlist: nil
        }
    }
}

@MainActor
public final class CloudKitLibraryProvider: LibraryProvider {
    public nonisolated let id = LibraryProviderID.cloudKit

    /// Everything — our own store has no primitive it lacks; the gaps are Apple's.
    public nonisolated let capabilities = LibraryCapabilities.all

    public enum Failure: Error, Equatable {
        /// No playlist with that local id belongs to this kid — deleted elsewhere.
        case unknownPlaylist(UUID)
    }

    private let kid: Kid
    private let context: ModelContext

    /// One provider per kid; switching kids builds a new one, like the engine.
    public init(kid: Kid, context: ModelContext) {
        self.kid = kid
        self.context = context
    }

    // MARK: - Favorites

    /// Most recent first; unreadable rows drop — a favorite isn't a permission.
    public func favorites() async throws -> [FavoriteRow] {
        kid.favorites
            .compactMap(\.row)
            .sorted { $0.favoritedAt > $1.favoritedAt }
    }

    public func addFavorite(
        _ id: CatalogID,
        displayName: String,
        artworkURL: URL?,
        origin: LibraryOrigin,
        ancestry: LibraryAncestry
    ) async throws {
        kid.addFavorite(
            id: id,
            displayName: displayName,
            artworkURL: artworkURL,
            origin: origin,
            ancestry: ancestry
        )
        try context.save()
    }

    /// Tops up lineage on favorites and entries, one save; failures just retry.
    public func recordAncestry(_ ancestry: [CatalogID: LibraryAncestry]) async throws {
        guard !ancestry.isEmpty else { return }

        var touched = false
        for record in kid.favorites {
            guard let id = record.itemID, let found = ancestry[id] else { continue }
            record.record(ancestry: found)
            touched = true
        }
        for playlist in kid.playlists {
            for entry in playlist.entries {
                guard let found = ancestry[entry.itemID] else { continue }
                entry.record(ancestry: found)
                touched = true
            }
        }

        guard touched else { return }
        try context.save()
    }

    public func removeFavorite(_ id: CatalogID) async throws {
        for record in kid.favorites.filter({ $0.itemID == id }) {
            context.delete(record)
        }
        kid.removeFavorite(id)
        try context.save()
    }

    // MARK: - The library index

    /// Sorted explicitly — SwiftData's relationship order isn't stable.
    public func indexed() async throws -> [LibraryIndexRow] {
        kid.libraryIndex
            .compactMap(\.row)
            .sorted { $0.catalogID.rawValue < $1.catalogID.rawValue }
    }

    /// Written as the walk finds things, so a closed app keeps what it got.
    public func appendIndex(from source: CatalogID, rows: [LibraryIndexRow]) async throws {
        guard !rows.isEmpty else { return }
        kid.appendIndex(from: source, with: rows)
        try context.save()
    }

    public func isIndexed(_ source: CatalogID) async -> Bool {
        kid.isIndexed(source)
    }

    public func removeIndex(from source: CatalogID) async throws {
        kid.removeIndex(from: source)
        try context.save()
    }

    // MARK: - Playlists

    /// Most recently touched first — the shelf shows what the kid is working on.
    public func playlists() async throws -> [PlaylistRow] {
        kid.playlists
            .map(\.row)
            .sorted { $0.modifiedAt > $1.modifiedAt }
    }

    public func createPlaylist(named name: String, songs: [LibrarySong]) async throws -> PlaylistRow
    {
        let record = kid.addPlaylist(named: name)
        for song in songs {
            record.addSong(
                id: song.item.id,
                displayName: song.item.title,
                artworkURL: song.item.artworkURL,
                ancestry: song.ancestry
            )
        }
        try context.save()
        return record.row
    }

    public func addToPlaylist(_ handle: PlaylistHandle, song: LibrarySong) async throws {
        let record = try playlist(handle)
        record.addSong(
            id: song.item.id,
            displayName: song.item.title,
            artworkURL: song.item.artworkURL,
            ancestry: song.ancestry
        )
        try context.save()
    }

    public func removeFromPlaylist(_ handle: PlaylistHandle, song: CatalogID) async throws {
        let record = try playlist(handle)
        for entry in record.entries.filter({ $0.itemID == song }) {
            context.delete(entry)
        }
        record.removeSong(song)
        try context.save()
    }

    public func reorderPlaylist(_ handle: PlaylistHandle, to order: [CatalogID]) async throws {
        try playlist(handle).reorder(to: order)
        try context.save()
    }

    public func renamePlaylist(_ handle: PlaylistHandle, to name: String) async throws {
        try playlist(handle).rename(to: name)
        try context.save()
    }

    public func deletePlaylist(_ handle: PlaylistHandle) async throws {
        let record = try playlist(handle)
        kid.removePlaylist(record)
        context.delete(record)
        try context.save()
    }

    /// Looked up by our stable `local` id, never Apple's, which may be absent.
    private func playlist(_ handle: PlaylistHandle) throws -> PlaylistRecord {
        guard let record = kid.playlistRecord(for: handle.local) else {
            throw Failure.unknownPlaylist(handle.local)
        }
        return record
    }
}

/// The real one; `UserDefaults`, because it's per-device and never synced.
public struct UserDefaultsDeviceLibraryState: DeviceLibraryStateStore {
    /// Namespaced so it reads as ours in a defaults dump.
    public static let key = "com.thegoldenmule.auralsandbox.deviceLibraryState"

    /// `UserDefaults` is thread-safe; `Sendable` just predates the annotation.
    nonisolated(unsafe) private let defaults: UserDefaults

    public init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    /// An unreadable value reads as unasked — never a throw, just one re-ask.
    public var state: DeviceLibraryState {
        guard
            let data = defaults.data(forKey: Self.key),
            let decoded = try? JSONDecoder().decode(DeviceLibraryState.self, from: data)
        else { return .unasked }
        return decoded
    }

    public func save(_ state: DeviceLibraryState) {
        guard let data = try? JSONEncoder().encode(state) else { return }
        defaults.set(data, forKey: Self.key)
    }
}

/// What a shelf draws; blocked items are kept, never dropped, for the report.
public struct LibraryResults: Sendable, Hashable {
    /// What the kid may see, in the order the shelf wants them.
    public let items: [CatalogItem]

    /// What the engine turned down, with its reason.
    public let blocked: [BlockedItem]

    /// False: a provisional verdict from stored copies; the recheck makes it safe.
    public let isResolved: Bool

    /// Rows not yet judged; zero means the read is complete, not merely paused.
    public let pending: Int

    public init(
        items: [CatalogItem],
        blocked: [BlockedItem],
        isResolved: Bool,
        pending: Int = 0
    ) {
        self.items = items
        self.blocked = blocked
        self.isResolved = isResolved
        self.pending = pending
    }

    public static let empty = LibraryResults(items: [], blocked: [], isResolved: true)
}

/// The kid's library across every provider; callers never name one.
@MainActor
public final class MergedLibrary {
    private let providers: [any LibraryProvider]
    private let catalog: any CatalogSource
    private let engine: PolicyEngine
    private let device: any DeviceLibraryStateStore

    public init(
        providers: [any LibraryProvider],
        catalog: any CatalogSource,
        engine: PolicyEngine,
        device: any DeviceLibraryStateStore
    ) {
        self.providers = providers
        self.catalog = catalog
        self.engine = engine
        self.device = device
    }

    // MARK: - Which providers are live

    /// `isSharedDevice` alone: false is both providers, true is CloudKit only.
    private var liveProviders: [any LibraryProvider] {
        let isShared = device.state.isSharedDevice
        return providers.filter { $0.id == .cloudKit || !isShared }
    }

    /// Read-live providers, narrower still: a read waits on onboarding's ask.
    private var readableProviders: [any LibraryProvider] {
        let hasBeenAsked = device.state.hasBeenAsked
        return liveProviders.filter { $0.id == .cloudKit || hasBeenAsked }
    }

    /// Providers written for this favorite; a missing capability skips it.
    private func writers(favoriting kind: ItemKind) -> [any LibraryProvider] {
        liveProviders.filter { $0.canFavorite(kind) }
    }

    // MARK: - Reading

    /// Every favorite the kid may see, filtered at render time, not pre-filtered.
    public func favorites() async throws -> LibraryResults {
        let rows = await mergedFavoriteRows()
        return await judge(
            rows.map { Stored(id: $0.catalogID, name: $0.displayName, artworkURL: $0.artworkURL) }
        )
    }

    /// Unfiltered favorite IDs; a blocked item never reaches a screen to ask.
    public func favoriteIDs() async -> Set<CatalogID> {
        Set(await mergedFavoriteRows().map(\.catalogID))
    }

    /// Every playlist, unfiltered — the engine has no opinion on a playlist.
    public func playlists() async -> [PlaylistRow] {

### Overview
A kid can **favorite** a song, an album or an artist, and can **build playlists** out of songs they are allowed to play.

### Disambiguation

The admin already reads a library: `LibrarySource`, the ISRC/UPC shortcut a parent uses to seed allow and deny lists from their own music. **That path is untouched and unrelated.** Nothing in this spec is called from the admin, and nothing in the admin is called from here.

### The store interface

We provide a simple store interface with multiple implementations for various scenarios.

- **CloudKit** — our own records in the existing named zone.
- **Apple Music** — the account's real library, through MusicKit.

`isSharedDevice`, in the admin, determines which if these is used.

- **On** — Read from and write to CloudKit only. This is so that the kid does not alter the adult's music library.
- **Off** — Read from both (resulting in union), write to both.

When both are live, a read is the union. CloudKit includes things Apple Music has no primitive for, and things that would not resolve.

**Drift between the two is not reconciled.** If an Apple Music write fails, the item still appears on the kid's shelf from CloudKit. Healing divergence is later work, to be done only if it turns out to matter.
